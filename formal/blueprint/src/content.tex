\chapter{The indexed table and split/union}

\begin{definition}[Indexed table]
  \label{def:table}\lean{Mensura.Table}\leanok
  A \emph{schema} is a column-name type $H$ with a per-column domain
  $\sigma : H \to \mathrm{Type}$.  A \emph{nested row} is a dependent
  function $(h : H) \to \mathrm{Option}\,(\sigma\,h)$, so each column
  carries its own, possibly missing, value.  An \emph{indexed table} over
  keys $K$ assigns to each key a multiset (bag) of nested rows; the empty
  bag is an absent row.
\end{definition}

\begin{definition}[Split]
  \label{def:split}\lean{Mensura.split}\uses{def:table}\leanok
  An indicator $s : K \to \mathrm{Bool}$ routes each key's whole bag to
  one of two output tables, leaving the other empty at that key
  (def:split of the book).
\end{definition}

\begin{definition}[Bind]
  \label{def:union}\lean{Mensura.union}\uses{def:table}\leanok
  The multiset union of two tables' bags at each key: the book's cell
  concatenation made order-free, commutative, associative, total, and
  bias-free (def:union).
\end{definition}

\begin{definition}[Disjoint tables]
  \label{def:disjoint}\lean{Mensura.Disjoint}\uses{def:table}\leanok
  Two tables are disjoint when at every key at least one of them is
  empty (def:disjoint-tables).
\end{definition}

\begin{theorem}[A split yields disjoint halves]
  \label{thm:split-disjoint}\lean{Mensura.split_disjoint}\leanok
  \uses{def:split, def:disjoint}
  The two halves of any split are disjoint.
\end{theorem}
\begin{proof}\leanok
  At each key the indicator sends the bag to one side only.
\end{proof}

\begin{theorem}[Bind undoes split]
  \label{thm:union-split}\lean{Mensura.union_split}\leanok
  \uses{def:split, def:union}
  For every indicator $s$ and table $T$, binding the two halves of
  $\mathrm{split}\,s\,T$ recovers $T$.
\end{theorem}
\begin{proof}\leanok
  Key-by-key case analysis on $s$.
\end{proof}

\begin{theorem}[Bind is commutative]
  \label{thm:union-comm}\lean{Mensura.union_comm}\uses{def:union}\leanok
  $\mathrm{union}\,T_0\,T_1 = \mathrm{union}\,T_1\,T_0$, unconditionally.
\end{theorem}
\begin{proof}\leanok
  Multiset union is commutative.
\end{proof}

\begin{theorem}[Bind is associative]
  \label{thm:union-assoc}\lean{Mensura.union_assoc}\uses{def:union}\leanok
  $\mathrm{union}$ is associative.
\end{theorem}
\begin{proof}\leanok
  Multiset union is associative.
\end{proof}

\begin{theorem}[Bind weakens disjointness]
  \label{thm:union-disjoint-iff}\lean{Mensura.union_disjoint_iff}\leanok
  \uses{def:union, def:disjoint}
  $\mathrm{union}\,T_0\,T_1$ is disjoint from $T_2$ iff both $T_0$ and
  $T_1$ are.  This backs the \texttt{union} propagation rule of the
  lineage system (\texttt{docs/language/08-lineage.md}).
\end{theorem}
\begin{proof}\leanok
  A multiset sum is empty exactly when both summands are.
\end{proof}

\begin{definition}[Split-invariance]
  \label{def:split-invariant}\lean{Mensura.SplitInvariant}\leanok
  \uses{def:union, def:disjoint}
  An operation $f$ is \emph{split-invariant} when it distributes over
  the union of disjoint tables (def:split-invariance).  This is the
  property Mensura tracks and enforces; disjointness is exactly what a
  split produces.
\end{definition}

\begin{definition}[Disjointness preservation]
  \label{def:preserves-disjoint}\lean{Mensura.PreservesDisjoint}\leanok
  \uses{def:disjoint}
  An operation preserves disjointness when it sends disjoint tables to
  disjoint tables.  This is the missing ingredient that makes
  split-invariance compositional.
\end{definition}

\begin{definition}[Split-safety]
  \label{def:splitsafe}\lean{Mensura.SplitSafe}\leanok
  \uses{def:split-invariant, def:preserves-disjoint}
  An operation is \emph{split-safe} when it is split-invariant and
  preserves disjointness: the class of operations safe to put in a
  pipeline between a split and a union.
\end{definition}

\begin{theorem}[Split-safe operations compose]
  \label{thm:splitsafe-comp}\lean{Mensura.SplitSafe.comp}\leanok
  \uses{def:splitsafe}
  The composition of split-safe operations is split-safe, so a whole
  pipeline of split-safe operations is split-invariant.
\end{theorem}
\begin{proof}\leanok
  The disjointness $f$ preserves is exactly what feeds $g$'s
  split-invariance.
\end{proof}

\begin{definition}[Minimality]
  \label{def:minimal}\lean{Mensura.Minimal}\uses{def:table}\leanok
  A table is \emph{minimal} when every nested row has at least one
  present column (the book's standing no-all-missing-row assumption).
\end{definition}

\chapter{The operations}

\begin{definition}[Row-wise flatMap]
  \label{def:flatMap}\lean{Mensura.flatMap}\uses{def:table}\leanok
  The single row-wise primitive: $\varphi\,k\,f$ sends a nested row to a
  bag of output rows (empty drops it, a singleton keeps or transforms
  it, several expand it).  It subsumes the book's def:selection,
  def:mutating, def:filtering, and the row-expanding direction of
  def:grouping.
\end{definition}

\begin{definition}[Left join]
  \label{def:leftjoin}\lean{Mensura.lookup}\uses{def:flatMap}\leanok
  def:left-join against a fixed right table on shared index columns:
  each present left row combines with every matching right row, or is
  kept once with missing right columns when there is no match.  A flatMap.
\end{definition}

\begin{definition}[Inner join]
  \label{def:innerjoin}\lean{Mensura.lookupTotal}\uses{def:flatMap}\leanok
  Like the left join, but an unmatched left row is dropped (def:join,
  fixed-right form).  A flatMap.
\end{definition}

\begin{definition}[Aggregate]
  \label{def:aggregate}\lean{Mensura.aggregate}\uses{def:table}\leanok
  def:aggregating: collapse each key's whole bag to a single row (empty
  stays empty).  The folder sees the entire bag, so this is not a flatMap.
\end{definition}

\begin{definition}[Ungroup]
  \label{def:promote}\lean{Mensura.promote}\uses{def:table}\leanok
  def:grouping: turn a distinguished total column into part of the key,
  refining $K$ to $K \times \beta$.
\end{definition}

\begin{definition}[Project]
  \label{def:demote}\lean{Mensura.demote}\uses{def:table}\leanok
  def:projection: drop a finite index component $D$ from the key,
  merging the bags of all dropped keys into one output key, each row
  tagged with its $d$.  This changes the observational unit.
\end{definition}

\begin{definition}[Tagged union]
  \label{def:tagged-union}\lean{Mensura.taggedBind}\leanok
  \uses{def:union, def:flatMap}
  def:tagged-union: union two tables while recording each row's source in
  a fresh column.
\end{definition}

\begin{definition}[Tagged split]
  \label{def:tagged-split}\lean{Mensura.taggedSplit}\uses{def:flatMap}\leanok
  def:tagged-split: recover the rows of one source, dropping the tag
  column.  A flatMap.
\end{definition}

\begin{theorem}[Tagged split inverts tagged union]
  \label{thm:tagged-inverse}\lean{Mensura.taggedSplit_taggedBind_left}\leanok
  \uses{def:tagged-union, def:tagged-split}
  With distinct tags, recovering source $s_0$ from
  $\mathrm{taggedBind}\,s_0\,s_1\,T_0\,T_1$ gives back $T_0$ (and
  symmetrically for $s_1$).
\end{theorem}
\begin{proof}\leanok
  $T_0$'s rows are kept and untagged; $T_1$'s carry the other tag and
  are filtered out.
\end{proof}

\chapter{Split-safety}

\begin{theorem}[Map is split-safe]
  \label{thm:flatMap-splitsafe}\lean{Mensura.flatMap_splitSafe}\leanok
  \uses{def:flatMap, def:splitsafe}
  Every flatMap is split-safe.
\end{theorem}
\begin{proof}\leanok
  A flatMap distributes over every union (multiset union distributes over
  multiset union), and it sends empty bags to empty bags, preserving
  disjointness.
\end{proof}

\begin{theorem}[Left join is split-safe]
  \label{thm:leftjoin-splitsafe}\lean{Mensura.lookup_splitSafe}\leanok
  \uses{def:leftjoin, def:splitsafe}
  The fixed-right left join is split-safe.
\end{theorem}
\begin{proof}\leanok
  It is a flatMap.
\end{proof}

\begin{theorem}[Inner join is split-safe]
  \label{thm:innerjoin-splitsafe}\lean{Mensura.lookupTotal_splitSafe}\leanok
  \uses{def:innerjoin, def:splitsafe}
  The fixed-right inner join is split-safe.  The book leaves the binary
  join open (see the open problems chapter); in the unary form the only
  effect is dropping unmatched left rows, a flatMap.
\end{theorem}
\begin{proof}\leanok
  It is a flatMap.
\end{proof}

\begin{theorem}[Ungroup is split-safe]
  \label{thm:promote-splitsafe}\lean{Mensura.promote_splitSafe}\leanok
  \uses{def:promote, def:splitsafe}
  Ungrouping is split-safe.
\end{theorem}
\begin{proof}\leanok
  Each output key reads a single input key through a multiset union.
\end{proof}

\begin{theorem}[Aggregate is split-safe]
  \label{thm:aggregate-splitsafe}\lean{Mensura.aggregate_splitSafe}\leanok
  \uses{def:aggregate, def:splitsafe}
  Aggregation is split-invariant (a split never divides a key's bag, so
  under disjointness folding the union equals folding the nonempty
  side) and preserves disjointness, hence is split-safe.
\end{theorem}
\begin{proof}\leanok
  At each key one summand is empty, and an empty bag aggregates to an
  empty bag.
\end{proof}

\begin{theorem}[Project does not preserve disjointness]
  \label{thm:demote-not-preservesdisjoint}
  \lean{Mensura.demote_not_preservesDisjoint}\leanok
  \uses{def:demote, def:preserves-disjoint}
  Two rows a split separates by the dropped index land on the same
  output key, so projection is \emph{not} split-safe: pipelines through
  it (such as aggregate after demote, the averaging case) can disagree
  with the full-table result.
\end{theorem}
\begin{proof}\leanok
  Counterexample splitting on the dropped Boolean index.
\end{proof}

\chapter{Reshape and the inverse pair}

\begin{definition}[Unpivot, reify variant]
  \label{def:unpivot}\lean{Mensura.unpivot}\uses{def:table}\leanok
  def:pivot-w2l: spread each name-column $n$ of a wide row into its own
  output key $(k, n)$, carrying that column's value; a missing cell
  becomes a long row holding a missing value.
\end{definition}

\begin{definition}[Pivot]
  \label{def:pivot}\lean{Mensura.pivot}\uses{def:table}\leanok
  def:pivot-l2w: gather the value at each name $n$ into one wide row per
  key $k$; a key whose names are all empty stays absent.
\end{definition}

\begin{definition}[Functional table]
  \label{def:functional}\lean{Mensura.Functional}\uses{def:table}\leanok
  A table is functional when every key holds at most one nested row (the
  book's ``card constant'' case).
\end{definition}

\begin{theorem}[Pivot is not split-invariant]
  \label{thm:pivot-not-splitinvariant}\lean{Mensura.pivot_not_splitInvariant}\leanok
  \uses{def:pivot, def:split-invariant}
  A split that separates the names of one key yields two complementary
  partial rows that union keeps apart.  This refines the book, whose
  pivot invariance relies on a cell-wise-merge union this total-row model
  deliberately does not have.
\end{theorem}
\begin{proof}\leanok
  Counterexample splitting a two-name key by name.
\end{proof}

\begin{definition}[Unpivot, drop variant]
  \label{def:unpivot-drop}\lean{Mensura.unpivotDrop}\uses{def:unpivot}\leanok
  ADR 0020: a missing cell yields \emph{no} long row, so value-missing
  in the wide table and row-absent in the long table carry the same
  information.  The long value column is total by construction.
\end{definition}

\begin{theorem}[The drop variant is split-safe]
  \label{thm:unpivotdrop-splitsafe}\lean{Mensura.unpivotDrop_splitSafe}\leanok
  \uses{def:unpivot-drop, def:splitsafe}
  Dropping a missing cell is a per-row decision, so the drop variant
  distributes over every union, hence is split-safe.
\end{theorem}
\begin{proof}\leanok
  Multiset union distributes over union.
\end{proof}

\begin{theorem}[Pivot inverts the drop variant]
  \label{thm:pivot-unpivotdrop}\lean{Mensura.pivot_unpivotDrop}\leanok
  \uses{def:pivot, def:unpivot-drop, def:functional, def:minimal}
  On functional, \emph{minimal} wide tables, pivoting the drop-variant
  unpivot recovers the table.  Minimality is load-bearing: an
  all-missing wide row would emit no long rows and its key would vanish.
\end{theorem}
\begin{proof}\leanok
  The substantive column supplies a present long row; the rest reads
  back cell by cell.
\end{proof}

\begin{theorem}[The drop variant inverts pivot]
  \label{thm:unpivotdrop-pivot}\lean{Mensura.unpivotDrop_pivot}\leanok
  \uses{def:pivot, def:unpivot-drop, def:functional, def:minimal}
  On functional, minimal long tables the round trip long-to-wide-to-long
  is the identity, with \emph{no completeness or saturation side
  condition}: a sparse long table round-trips as it is.  This direction
  is not statable for the reify variant; it is the mechanized content of
  ADR 0020's inverse contract.
\end{theorem}
\begin{proof}\leanok
  An absent $(k, n)$ row pivots to a missing cell, which the drop
  variant sends back to an absent row.
\end{proof}

\begin{theorem}[Project inverts promote]
  \label{thm:demote-promote}\lean{Mensura.demote_promote}\leanok
  \uses{def:promote, def:demote}
  ADR 0024: when the promoted column is total (exactly the checker's
  \texttt{extend\_key} gate, which would otherwise drop rows at
  \texttt{promote}), demoting the component again restores the table
  exactly.  The \texttt{extend\_key}/\texttt{shrink\_key} round trip is
  the identity, hence preserves every tracked property at once.
\end{theorem}
\begin{proof}\leanok
  Push the tagging flatMap into the fiber union, exchange the finite sum with
  the union, and each row collapses to the singleton of itself: the tag
  it receives is the key component it was filed under.
\end{proof}

\begin{theorem}[Ungroup inverts demote]
  \label{thm:promote-demote}\lean{Mensura.promote_demote}\leanok
  \uses{def:promote, def:demote}
  ADR 0024, with no side condition: \texttt{demote} tags every demoted
  row with its own key component, so the demoted column is total by
  construction and re-promoting it files every row back under exactly
  the key it came from.
\end{theorem}
\begin{proof}\leanok
  Distribute the union over the fiber sum; at each fine key the tag
  matches exactly its own component, selecting the original fiber.
\end{proof}

\chapter{The rectangle fact}

\begin{definition}[Total table]
  \label{def:total}\lean{Mensura.Total}\uses{def:table}\leanok
  Every cell of every present row is known: the table-level reading of
  ``all columns total'' (ADR 0010's default).
\end{definition}

\begin{definition}[Exhaustive long table]
  \label{def:exhaustive}\lean{Mensura.Exhaustive}\uses{def:table}\leanok
  ADR 0020's rectangle fact: every residual key present in a long table
  carries a row for \emph{every} name.  The reference is the name type's
  whole domain, not a population, which distinguishes it from the
  population-relative \texttt{complete\_over}.
\end{definition}

\begin{theorem}[The drop variant's output is minimal]
  \label{thm:unpivotdrop-minimal}\lean{Mensura.unpivotDrop_minimal}\leanok
  \uses{def:unpivot-drop, def:minimal}
  Every long row the drop variant emits carries a known value.
\end{theorem}
\begin{proof}\leanok
  By construction of the drop.
\end{proof}

\begin{theorem}[The drop variant establishes the rectangle]
  \label{thm:unpivotdrop-exhaustive}\lean{Mensura.unpivotDrop_exhaustive}\leanok
  \uses{def:unpivot-drop, def:total, def:exhaustive}
  On a total wide table, each source row emits one long row per name, so
  a residual key present for one name is present for all.
\end{theorem}
\begin{proof}\leanok
  Totality supplies the value at every name.
\end{proof}

\begin{theorem}[The totality upgrade]
  \label{thm:pivot-total}\lean{Mensura.pivot_total_of_exhaustive}\leanok
  \uses{def:pivot, def:exhaustive, def:minimal, def:total}
  Pivoting an exhaustive, minimal long table yields only known cells:
  exhaustiveness supplies the row at every $(k, n)$ and minimality the
  value in it.  No population-relative completeness fact appears.
\end{theorem}
\begin{proof}\leanok
  Read the guarded cell through the present fiber.
\end{proof}

\begin{theorem}[A single-variant name axis is exhaustive]
  \label{thm:exhaustive-subsingleton}
  \lean{Mensura.exhaustive_of_subsingleton}\leanok
  \uses{def:exhaustive}
  When the name type has at most one element, a present residual key has
  a row for some name and there is only one name to have, so the table is
  exhaustive trivially.  This backs the checker's single-variant
  refinement of the totality upgrade.
\end{theorem}
\begin{proof}\leanok
  The unique name equals any name by subsingleton elimination.
\end{proof}

\begin{theorem}[A non-dropping flatMap preserves the rectangle]
  \label{thm:flatMap-exhaustive}\lean{Mensura.flatMap_exhaustive}\leanok
  \uses{def:flatMap, def:exhaustive}
  A flatMap whose body emits at least one output row per input row preserves
  exhaustiveness.  A dropping flatMap (a filter) is excluded, and rightly
  so: it can empty one name of a fiber.
\end{theorem}
\begin{proof}\leanok
  Presence is preserved in both directions.
\end{proof}

\begin{theorem}[Left join preserves the rectangle]
  \label{thm:leftjoin-exhaustive}\lean{Mensura.lookup_exhaustive}\leanok
  \uses{def:leftjoin, def:exhaustive, thm:flatMap-exhaustive}
  The left join's body emits the unmatched row with missing right
  columns rather than dropping it, so it is a non-dropping flatMap.
\end{theorem}
\begin{proof}\leanok
  Instance of the non-dropping flatMap lemma.
\end{proof}

\begin{theorem}[Aggregate preserves the rectangle]
  \label{thm:aggregate-exhaustive}\lean{Mensura.aggregate_exhaustive}\leanok
  \uses{def:aggregate, def:exhaustive}
  Aggregation collapses each present fiber to one row and leaves absent
  fibers absent, so presence is untouched.
\end{theorem}
\begin{proof}\leanok
  Presence is preserved in both directions.
\end{proof}

\begin{theorem}[Bind preserves the rectangle]
  \label{thm:union-exhaustive}\lean{Mensura.union_exhaustive}\leanok
  \uses{def:union, def:exhaustive}
  The union of two exhaustive tables is exhaustive: a fiber present in
  the union is present on one side, whose fullness fills the union.
\end{theorem}
\begin{proof}\leanok
  Presence in the union decomposes over the summands.
\end{proof}

\begin{theorem}[Split destroys the rectangle]
  \label{thm:split-not-exhaustive}\lean{Mensura.split_not_exhaustive}\leanok
  \uses{def:split, def:exhaustive}
  A predicate that reads the name axis cuts a fiber, leaving one side
  with some names of a key but not others: ADR 0020's ``destroyed by
  split'' row, and the honest price of keeping the name in the key.
\end{theorem}
\begin{proof}\leanok
  Counterexample splitting on the name axis.
\end{proof}

\chapter{Safe completeness}

\begin{definition}[Fiber flatMap]
  \label{def:fibermap}\lean{Mensura.fiberMap}\uses{def:table}\leanok
  The output bag at each key is a function of the input bag at the
  \emph{same} key and nothing else.  \texttt{flatMap} and \texttt{aggregate}
  are both fiber maps.
\end{definition}

\begin{definition}[Strict fiber action]
  \label{def:strict}\lean{Mensura.Strict}\uses{def:fibermap}\leanok
  A fiber action is strict when it sends the empty bag to the empty bag:
  it does not fabricate rows for an absent key.
\end{definition}

\begin{definition}[Key-locality]
  \label{def:keylocal}\lean{Mensura.KeyLocal}\uses{def:table}\leanok
  A key-preserving operation is key-local when each output key depends
  only on the input at that same key.
\end{definition}

\begin{theorem}[A strict fiber flatMap is split-safe]
  \label{thm:fibermap-splitsafe}\lean{Mensura.fiberMap_splitSafe}\leanok
  \uses{def:fibermap, def:strict, def:splitsafe}
  Every strict fiber flatMap is split-safe.
\end{theorem}
\begin{proof}\leanok
  Under disjointness one summand is empty at each key, and strictness
  makes folding the union the same as folding the nonempty side.
\end{proof}

\begin{theorem}[Safe completeness, key-preserving fragment]
  \label{thm:fibermap-iff}\lean{Mensura.splitInvariant_keyLocal_iff_fiberMap}\leanok
  \uses{def:fibermap, def:strict, def:keylocal, thm:fibermap-splitsafe}
  A key-preserving operation is split-invariant and key-local \emph{iff}
  it is a strict fiber flatMap.  The split-invariant, key-local
  transformations are exactly the strict fiber maps: nothing safe is
  missing and nothing unsafe sneaks in.
\end{theorem}
\begin{proof}\leanok
  Forward: run the operation on point tables to witness its fiber
  action; key-locality makes the representation exact and
  split-invariance makes it strict.  Backward: the strict fiber flatMap
  results.
\end{proof}

\begin{theorem}[A presence-preserving fiber flatMap preserves the rectangle]
  \label{thm:fibermap-exhaustive}\lean{Mensura.fiberMap_exhaustive}\leanok
  \uses{def:fibermap, def:strict, def:exhaustive, thm:aggregate-exhaustive}
  A strict fiber action that never empties a present fiber preserves
  exhaustiveness: strictness keeps absent fibers absent and the
  no-emptying hypothesis keeps present fibers present, so full fibers
  stay full.  This is the universal form of which
  \ref{thm:aggregate-exhaustive} and \ref{thm:flatMap-exhaustive} are the two
  principal cases.
\end{theorem}
\begin{proof}\leanok
  Presence is preserved in both directions.
\end{proof}

\begin{definition}[Reindexing fiber flatMap]
  \label{def:reindexmap}\lean{Mensura.reindexMap}\uses{def:fibermap}\leanok
  Along $r : K' \to K$, the output bag at key $k'$ is a function of the
  input bag at the single key $r\,k'$.  The fiber flatMap is the $r = id$
  case; \texttt{promote} and \texttt{unpivot} are $r = \pi_1$ cases.
\end{definition}

\begin{definition}[Locality along a reindexing]
  \label{def:reindexlocal}\lean{Mensura.ReindexLocal}\uses{def:keylocal}\leanok
  Each output key $k'$ depends only on the input at $r\,k'$; for
  $r = id$ this is exactly key-locality.
\end{definition}

\begin{theorem}[Safe completeness, key-changing fragment]
  \label{thm:reindex-iff}
  \lean{Mensura.splitInvariant_reindexLocal_iff_reindexMap}\leanok
  \uses{def:reindexmap, def:reindexlocal, def:strict}
  Along any $r : K' \to K$, an operation is split-invariant and
  $r$-local \emph{iff} it is a strict reindexing fiber flatMap over $r$: the
  full generalization of the key-preserving characterization.
\end{theorem}
\begin{proof}\leanok
  Same witnessing argument, run at the reindexed point table.
\end{proof}

\begin{definition}[Attribute pivot]
  \label{def:pivotattr}\lean{Mensura.pivotAttr}\leanok
  \uses{def:fibermap, def:pivot}
  The relaxed pivot spreading a \emph{non-index} name column: the long
  table is indexed by the residual key only, with (value, name) pairs in
  the bag, so pivoting reads only one key's bag.
\end{definition}

\begin{theorem}[The attribute pivot is split-safe]
  \label{thm:pivotattr-splitsafe}\lean{Mensura.pivotAttr_splitSafe}\leanok
  \uses{def:pivotattr, thm:fibermap-splitsafe}
  In contrast to the index-column pivot
  (\ref{thm:pivot-not-splitinvariant}): the formal counterpart of the
  book's relaxed pivot, safe when the name is an attribute, unsafe when
  it is an index.
\end{theorem}
\begin{proof}\leanok
  It is a strict fiber flatMap.
\end{proof}

\section{The monoid-parameterized fold}

This section is ADR 0029's Stage 1, the gate on the surface \texttt{fold}
(ADR 0031).  Its subject is the \emph{subclass} of \ref{def:aggregate}
whose combiner is associative and commutative: those two laws are what make
the result independent of the bag's arbitrary order, and they are laws no
checker can verify on a user-supplied lambda, which is why the surface draws
its combiner from a closed table whose rows are exactly the instances
admitted here.

\begin{definition}[Bag fold over a commutative monoid]
  \label{def:foldbag}\lean{Mensura.foldBag}\uses{def:table}\leanok
  Fold a bag with an associative-commutative operation from an identity
  accumulator.  The multiset quotient makes this well defined without
  choosing an order: associativity and commutativity are precisely the
  hypotheses the fold demands.
\end{definition}

\begin{theorem}[The shard lemma]
  \label{thm:foldbag-shards}\lean{Mensura.foldBag_shards}\uses{def:foldbag}\leanok
  Folding each shard and then folding the partial results equals folding
  the union, for arbitrarily many shards including empty ones, provided
  the accumulator is the combiner's identity.  This is what licenses
  partial and parallel folding.
\end{theorem}
\begin{proof}\leanok
  Induction on the shard multiset, with the two-shard case
  (\texttt{foldBag\_add}) as the step; an empty shard contributes the
  identity, which is why it is harmless.
\end{proof}

\begin{theorem}[A non-identity seed breaks the shard lemma]
  \label{thm:foldbag-seed-counterexample}
  \lean{Mensura.foldBag_add_seed_counterexample}\uses{thm:foldbag-shards}\leanok
  A user-supplied seed that is not the combiner's identity is counted once
  per shard, so the shard lemma fails: witnessed on addition with seed one
  over two empty shards.  This is why the identity lives in the
  accumulator and the surface has no seed argument (ADR 0029
  Alternative 3).
\end{theorem}
\begin{proof}\leanok
  Evaluate both sides.
\end{proof}

\begin{definition}[The optional completion]
  \label{def:optionlift}\lean{Mensura.optionLift}\leanok
  Extend a binary operation to \texttt{Option}, letting the absent value
  act as an identity the domain itself does not contain.  It is
  associative and commutative whenever the operation is, so it feeds the
  same fold.
\end{definition}

\begin{definition}[Bag fold over a commutative semigroup]
  \label{def:foldbag-opt}\lean{Mensura.foldBagOpt}\leanok
  \uses{def:foldbag, def:optionlift}
  The fold for a combiner with \emph{no} identity in its own domain
  (binary minimum and maximum): the empty bag folds to the absent value.
  There is no smallest element of nothing, so the absence is the honest
  answer rather than a fabricated sentinel.
\end{definition}

\begin{theorem}[The presence lemma]
  \label{thm:foldbag-opt-issome}\lean{Mensura.foldBagOpt_isSome_of_ne_zero}\leanok
  \uses{def:foldbag-opt}
  A non-empty bag folds to a present value, and (conversely) only the
  empty bag folds to the absent one.  This licenses the total surface type
  of the identity-free reductions wherever the checker has proved the
  fiber present: the option is an artefact of the empty case and never a
  data-dependent surprise.
\end{theorem}
\begin{proof}\leanok
  Induction on the bag, splitting on whether the tail is empty.
\end{proof}

\begin{definition}[The fold as a fiber action]
  \label{def:foldfiber}\lean{Mensura.foldFiber}\leanok
  \uses{def:foldbag, def:fibermap}
  Read each row with the mapper, fold the results with the combiner, and
  write the answer into the single output row.  An absent fiber stays
  absent.
\end{definition}

\begin{theorem}[A monoid fold is an aggregation]
  \label{thm:foldfiber-eq-aggregate}\lean{Mensura.foldFiber_eq_aggregate}\leanok
  \uses{def:foldfiber, def:aggregate}
  The parameterized fold is the well-behaved subclass of the algebra's
  whole-bag operation, not a new operation.  Since \ref{def:aggregate}
  takes an \emph{arbitrary} whole-bag function, this carves out a
  subclass rather than generalizing.
\end{theorem}
\begin{proof}\leanok
  Both sides compute the same bag at each key.
\end{proof}

\begin{theorem}[A monoid fold is split-safe]
  \label{thm:foldfiber-splitsafe}\lean{Mensura.foldFiber_splitSafe}\leanok
  \uses{def:foldfiber, thm:fibermap-splitsafe}
  Hence it composes into pipelines that stay split-invariant.  A split
  routes a key's \emph{whole} bag to one side (\ref{def:split}), so the
  bag a fold sees is never torn, which is why the reducing group
  transform is safe (ADR 0023).
\end{theorem}
\begin{proof}\leanok
  It is a strict fiber flatMap.
\end{proof}

\begin{theorem}[A monoid fold preserves the rectangle]
  \label{thm:foldfiber-exhaustive}\lean{Mensura.foldFiber_exhaustive}\leanok
  \uses{def:foldfiber, thm:fibermap-exhaustive}
  A present fiber folds to exactly one row, so no fiber loses its
  presence.
\end{theorem}
\begin{proof}\leanok
  The no-emptying hypothesis of \ref{thm:fibermap-exhaustive} holds by
  construction.
\end{proof}

\section{The arranged structure}

This section is ADR 0029's Stage 2, the gate on the surface \texttt{scan},
\texttt{prescan}, \texttt{desc}, and the \texttt{series} module (ADR 0031
Decision 7).  \ref{def:table} fixes a fiber's content as a multiset, and the
core module argues \emph{for} that choice: the order of nested rows is
``arbitrary but fixed'', so the model declines to assert an order it never
uses.  That argument is why the ordered verbs need new structure, and the
structure added here does not weaken the substrate: content stays a
multiset, and an arrangement is a \emph{relation} between it and a list of
the same elements.

\begin{definition}[Arranged tables and window operations]
  \label{def:arranged}\lean{Mensura.IsArrangement}\uses{def:table}\leanok
  Order-dependent verbs (lag, cumulative sum, rank) are not bag
  operations: lift the content from the bag monad to the list monad, as in
  the book's ``grouped and arranged'' section.  A list arranges a bag by a
  key when it holds exactly the bag's elements and its keys are
  non-decreasing.

  Stated as a relation rather than as a chosen sort, because existence and
  uniqueness have different hypotheses (\ref{thm:arrange-exists},
  \ref{thm:arrange-unique}) and a sort conflates them.

  Split-invariance is a hypothesis to pin, not a contradiction to resolve.
  A \ref{def:split} routes each key's \emph{whole} multiset to one side, so
  a fiber is never torn and an arranged verb sorts an intact bag: the
  results agree whichever side the key lands on
  (\ref{thm:scanfiber-splitsafe}).  What is deliberately \emph{not}
  split-invariant is lifting to the list monad in general, where an order
  \emph{across} keys would be asserted; these operations, under this split,
  are safe.  Determinism is the separate obligation, discharged by
  \ref{thm:arrange-unique}.
\end{definition}

\begin{theorem}[Every fiber can be arranged]
  \label{thm:arrange-exists}\lean{Mensura.exists_isArrangement}\leanok
  \uses{def:arranged}
  With no hypothesis on the key.  Merge sort at the key-induced order
  supplies the witness, and the pairwise property it guarantees needs only
  totality and transitivity, neither of which mentions ties.
\end{theorem}
\begin{proof}\leanok
  Induction on the quotient, then merge sort's permutation and pairwise
  lemmas.
\end{proof}

\begin{theorem}[Tier 1 determinism: a tie-free key arranges uniquely]
  \label{thm:arrange-unique}\lean{Mensura.IsArrangement.unique}\leanok
  \uses{def:arranged, def:functional}
  A key injective \emph{on the fiber} admits exactly one arrangement.  This
  is ADR 0029 Decision 11's first tier, and it is the reason the ordered
  surface needs a totality obligation the unordered one does not: two rows
  with equal keys are interchangeable, so without injectivity the
  intermediate values of a scan are not determined.

  The standard sort operation cannot state this.  It requires antisymmetry
  as a typeclass instance on the element type, which for a key-induced
  order is global injectivity of the key; tier 1 supplies only the
  per-fiber fact, and a per-fiber fact yields no global instance.  The
  proof therefore uses the permutation lemma whose antisymmetry argument is
  a pointwise hypothesis about the members.
\end{theorem}
\begin{proof}\leanok
  The two arrangements are permutations of each other; discharge
  antisymmetry pointwise from injectivity and the order's own antisymmetry.
\end{proof}

\begin{definition}[The inclusive and exclusive scans]
  \label{def:scanbag}\lean{Mensura.scanBag}\uses{def:arranged}\leanok
  The inclusive scan's position $i$ carries the fold of elements $1..i$;
  the exclusive scan's (\texttt{prescanBag}, after Blelloch) carries
  $1..i-1$, so its first entry folds the empty prefix.  Both are slices of
  one list scan, which is natively exclusive plus a final total: drop the
  seed for the inclusive form, drop the total for the exclusive one.

  Hence the first entry of an exclusive scan is the combiner's identity, or
  \emph{absent} for a combiner that has none.  The missing first row of a
  lag is therefore not designed; it falls out of the identity-and-emptiness
  rule that already governs \ref{def:foldbag-opt}.
\end{definition}

\begin{theorem}[Scan/fold coherence]
  \label{thm:scan-fold-coherence}\lean{Mensura.scanl_getLast_eq_foldBag}\leanok
  \uses{def:scanbag, def:foldbag}
  The last entry of an inclusive scan over \emph{any} arrangement equals
  the bag's fold.  This is what makes ``same combiner, two variants'' a
  theorem rather than a slogan: the ordered variant refines the unordered
  one instead of departing from it, which is why one closed combiner table
  serves both.

  Note what it does not need: injectivity.  Ties permute the arrangement,
  but an associative-commutative combiner cannot observe the difference, so
  the total is determined even when the intermediates are not.  That
  asymmetry is exactly why a fold needs no order and a scan does.  The
  claim is also per combiner class: for the associative-only tacks no fold
  exists to cohere with, which is the formal content of the surface rule
  admitting them under a scan only.
\end{theorem}
\begin{proof}\leanok
  The last element of a list scan is the left fold; transport along the
  arrangement's multiset equality and use the fold's commutativity.
\end{proof}

\begin{theorem}[The prefix decomposition]
  \label{thm:scan-prefix-decomp}\lean{Mensura.scanl_append_decomp}\leanok
  \uses{def:scanbag}
  Scanning a concatenation is the first part's scan followed by the second
  part's scan seeded at the first part's total, which is what licenses a
  parallel scan.  Stage 2's analogue of \ref{thm:foldbag-shards}, and
  unlike it this needs no laws at all, because a list scan asserts no
  order-independence to protect.
\end{theorem}
\begin{proof}\leanok
  The list scan's append lemma.
\end{proof}

\begin{definition}[The scan as a fiber action]
  \label{def:scanfiber}\lean{Mensura.scanFiber}\leanok
  \uses{def:scanbag, def:fibermap}
  Arrange the fiber by the key, scan the mapped elements, and write one
  output row per input row.  Unlike \ref{def:foldfiber} this is a
  \emph{window} shape: the output bag has the input's cardinality, which is
  the second group-transform shape \ref{thm:fibermap-exhaustive} already
  anticipated.
\end{definition}

\begin{theorem}[A scan is split-safe]
  \label{thm:scanfiber-splitsafe}\lean{Mensura.scanFiber_splitSafe}\leanok
  \uses{def:scanfiber, thm:fibermap-splitsafe}
  The lemma \ref{def:arranged} promises.  A scan is a strict fiber
  transform, so it inherits safety unchanged: the arrangement is computed
  inside one key's fiber, and a split never tears a fiber.
\end{theorem}
\begin{proof}\leanok
  Strictness, then the fiber-transform safety lemma.  The empty bag
  arranges to the empty list, whose scan is empty.
\end{proof}

\begin{theorem}[A scan preserves the rectangle]
  \label{thm:scanfiber-exhaustive}\lean{Mensura.scanFiber_exhaustive}\leanok
  \uses{def:scanfiber, thm:fibermap-exhaustive}
  One output row per input row, so no fiber loses its presence.
\end{theorem}
\begin{proof}\leanok
  A non-empty fiber arranges to a non-empty list, and the scan preserves
  its length.
\end{proof}

\begin{theorem}[Functional discharges tier 1, vacuously]
  \label{thm:keyinjon-of-functional}\lean{Mensura.keyInjOn_of_functional}\leanok
  \uses{thm:arrange-unique, def:functional}
  A functional table's fibers hold at most one row, so no key can tie.
  Recorded because ADR 0029 asks the determinism hypothesis to compose with
  functionality rather than restate it, and it does; but the composition is
  \emph{vacuous}, since a one-row fiber makes any scan a one-element list.
  So it is a compatibility bridge (a singleton-cardinality store needs no
  tie ceremony), not evidence that a scan over a real bag is deterministic.
  The substantive hypothesis is \ref{thm:keyinjon-demote-tag}.
\end{theorem}
\begin{proof}\leanok
  At cardinality at most one there are no two distinct members.
\end{proof}

\begin{theorem}[A projected key component is injective on every fiber]
  \label{thm:keyinjon-demote-tag}\lean{Mensura.keyInjOn_demote_tag}\leanok
  \uses{thm:arrange-unique, def:demote, def:functional}
  The \emph{substantive} tier 1 discharge, and the theorem a checker rule may
  cite.  \ref{def:demote} merges the rows of every key $(k,d)$ into the single
  output key $k$, tagging each with its own $d$.  So if the input is
  functional, at most one row per $(k,d)$, then each output fiber holds at most
  one row per tag value, and the tag column is injective on the fiber.

  Unlike \ref{thm:keyinjon-of-functional} this is not vacuous: the fiber may
  hold many rows, and what is bounded is how many share a tag.  It is also the
  common shape of a window (a history keyed by entity and time, projected to
  the entity, scanned in time order), and the surface tracks exactly this fact
  as a grading over the flat table, carried unchanged through the projection
  because a grading does not depend on which columns currently form the key
  (ADR 0024).
\end{theorem}
\begin{proof}\leanok
  Two rows of one output fiber agreeing on the tag came from one input key,
  which holds at most one row, so they are the same row.
\end{proof}

\chapter{Physical dimensions}

This chapter is standalone: it backs ADR 0026's dimensional-unit model and
neither depends on nor changes the table algebra above.

\begin{definition}[Base dimension]
  \label{def:base-dimension}\lean{Mensura.Units.Base}\leanok
  The seven SI base dimensions (time, length, mass, current, temperature,
  amount of substance, and luminous intensity), a finite type with decidable
  equality.
\end{definition}

\begin{definition}[Dimension]
  \label{def:dimension}\lean{Mensura.Units.Dimension}\uses{def:base-dimension}\leanok
  A physical dimension is an element of the free abelian group over the seven
  base dimensions: an integer exponent vector, written multiplicatively so
  that dimensions multiply while their exponents add, with the dimensionless
  quantity as the identity.  The group is commutative (so dimension
  multiplication, division, and integer powers are well defined) and its
  equality is decidable (so a unit mismatch is a decision procedure).  These
  are the two facts the expression checker relies on (ADR 0026).
\end{definition}

\begin{definition}[Exponent vector]
  \label{def:dimension-exponents}\lean{Mensura.Units.Dimension.exponents}\leanok
  \uses{def:dimension}
  The canonical form of a dimension: its integer exponent on each base
  dimension.  A base dimension has exponent one on itself and zero elsewhere;
  a product adds exponents, an inverse negates them, a quotient subtracts.
\end{definition}

\begin{theorem}[The seven axes are independent]
  \label{thm:dimension-base-injective}
  \lean{Mensura.Units.Dimension.base_injective}\leanok
  \uses{def:base-dimension, def:dimension-exponents}
  The seven base dimensions are pairwise distinct, so the group is genuinely
  of rank seven and no two axes collapse.
\end{theorem}
\begin{proof}\leanok
  Distinct base dimensions differ in their exponent vectors: each is one on
  its own axis and zero on the others.
\end{proof}

\chapter{Temporal torsors}

This chapter backs ADR 0036's torsor arithmetic for temporal point
domains (decision 9).  The abstract laws are stated over any additive
torsor, so neither a later exact backing (ADR 0026 decision 9) nor the
deferred civil instantiation (\texttt{diff(date)}) reopens them; the
concrete instantiation is the absolute point on the exact millisecond
grid.  The order-compatibility result is the lemma ADR 0037's window
gates (\texttt{closedWindow\_stable}, the interval test) rest on.

\begin{theorem}[The torsor laws]
  \label{thm:torsor-laws}
  \lean{Mensura.Units.Torsor.translate_vsub, Mensura.Units.Torsor.vsub_add_vsub,
    Mensura.Units.Torsor.vsub_self, Mensura.Units.Torsor.vsub_unique}\leanok
  For a point domain $P$ with difference group $G$ acting freely and
  transitively, the checker's laws follow from the structure rather than
  being stipulated: $t + (u - t) = u$, $(t - u) + (u - v) = t - v$,
  $t - t = 0$, and $u - t$ is the unique translation carrying $t$ to $u$.
\end{theorem}
\begin{proof}\leanok
  Each is a one-line consequence of the torsor axioms (the cancellation
  pair), instantiated from mathlib's \texttt{AddTorsor}.
\end{proof}

\begin{definition}[Duration]
  \label{def:duration}\lean{Mensura.Units.Duration}\leanok
  \uses{def:dimension}
  A duration on the exact millisecond grid: an integer count of
  milliseconds, forming a linearly ordered abelian group.  Its physical
  dimension is the \texttt{time} axis of the dimension group, which is how
  the torsor extends ADR 0026's structure rather than sitting beside it.
  The checker-level difference type is \texttt{time[real]}; the bridge to
  the binary64 backing is ADR 0036 decision 6's round-trip bound, an
  implementation obligation rather than a formal target.
\end{definition}

\begin{definition}[Instant]
  \label{def:instant}\lean{Mensura.Units.Instant}\leanok
  An absolute temporal point: a count of milliseconds from the epoch
  (ADR 0036 decision 5), linearly ordered by that count.  A structure
  rather than an integer synonym, so no numeric operation is available on
  points themselves (decision 3): every operation flows through the torsor
  interface.
\end{definition}

\begin{theorem}[Instant is a torsor over the millisecond durations]
  \label{thm:instant-torsor}
  \lean{Mensura.Units.Instant.addTorsor}\leanok
  \uses{def:duration, def:instant, thm:torsor-laws}
  Translation and difference on the exact millisecond grid satisfy the
  torsor axioms, so every law of \ref{thm:torsor-laws} holds at
  \texttt{instant}, with no assumption that the backing is real
  (ADR 0036 decision 9, item 2).
\end{theorem}
\begin{proof}\leanok
  Both cancellation axioms reduce to integer arithmetic on the
  millisecond counts.
\end{proof}

\begin{theorem}[Order compatibility]
  \label{thm:instant-order-compat}
  \lean{Mensura.Units.Instant.lt_vadd_of_pos,
    Mensura.Units.Instant.strictMono_vadd,
    Mensura.Units.Instant.vadd_lt_vadd_iff_left}\leanok
  \uses{def:duration, def:instant, thm:instant-torsor}
  Translation by a positive duration is strictly increasing, translation
  by a fixed duration is strictly monotone in the point, and translation
  of a fixed point is strictly monotone in the duration (ADR 0036
  decision 9, item 3).  This is the lemma ADR 0037's interval test
  $w \le p < w + \mathit{size}$ and its \texttt{closedWindow\_stable}
  gate rest on, so this chapter lands before the window lemmas.
\end{theorem}
\begin{proof}\leanok
  Each statement reduces to the corresponding fact about integer
  addition and order on the millisecond counts.
\end{proof}
